17# What 0.1 + 0.2 comes to in a double, against a then that says 0.3. 

18GIVEN = f'''@prefix t: <{T}> . 

19 @prefix xsd: <{X}> . 

20 t:a t:pred "0.30000000000000004"^^xsd:double ; t:label "1" .''' 

21 

22INLINE_THEN = """[ a must:TableDataset ; {extra} 

23 must:hasRow [ must:hasBinding [ must:variable "s" ; must:boundValue test-data:a ] , 

24 [ must:variable "o" ; must:boundValue "0.3"^^xsd:double ] ] ]""" 

25

27def run(then_ttl: str, spec_extra: str = "", store_tolerance: Tolerance = None, predicate: str = "pred"): 

28 given = Graph().parse(data=GIVEN, format="ttl") 

29 spec = f""" 

30 @prefix must: <https://mustrd.org/model/> . 

31 @prefix xsd: <{X}> . 

32 @prefix test-data: <{T}> . 

33 test-data:my_first_spec a must:TestSpec ; {spec_extra} 

34 must:when [ a must:TextSparqlSource ; 

35 must:queryText "select ?s ?o {{ ?s <{T}{predicate}> ?o }}" ; 

36 must:queryType must:SelectSparql ] ; 

37 must:then {then_ttl} . 

38 """ 

39 triple_store = {"type": TRIPLESTORE.RdfLib, "given": given} 

40 if store_tolerance: 

41 triple_store["tolerance"] = store_tolerance 

42 spec_uri = TEST_DATA.my_first_spec 

43 spec_graph = parse_spec(spec, spec_uri, __name__) 

44 run_config = {'spec_path': ""} 

45 when = parse_spec_component(subject=spec_uri, predicate=MUST.when, spec_graph=spec_graph, 

46 run_config=run_config, mustrd_triple_store=triple_store) 

47 then = parse_spec_component(subject=spec_uri, predicate=MUST.then, spec_graph=spec_graph, 

48 run_config=run_config, mustrd_triple_store=triple_store) 

49 spec = Specification(spec_uri, triple_store, given, when, then, 

50 tolerance=read_tolerance(spec_graph, spec_uri)) 

51 return check_result(spec, run_when(spec_uri, triple_store, when[0]))
