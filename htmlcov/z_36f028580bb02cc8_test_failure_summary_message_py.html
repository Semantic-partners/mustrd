1"""The one-line summary above a table diff. 

2 

3On a value- or datatype-only mismatch the expected and actual shapes are 

4identical, so "Expected 1 row(s) and 1 column(s), got 1 row(s) and 1 column(s)" 

5reads as though nothing is wrong — and on a wide result the column that actually 

6differs can sit off the right-hand edge of the diff below it. The summary names 

7the differing columns in exactly that case, and stays out of the way when the 

8shapes already carry the news. 

9 

10https://github.com/Semantic-partners/mustrd/issues/240 

11""" 

12 

13import pandas 

14import pytest

16from mustrd.mustrd import build_summary_message, describe_differing_columns

18XSD = "http://www.w3.org/2001/XMLSchema#"

21def diff_of(expected: dict, actual: dict) -> pandas.DataFrame: 

22 """The (column, expected|actual) MultiIndex frame mustrd compares with.""" 

23 return pandas.DataFrame(expected).compare( 

24 pandas.DataFrame(actual), result_names=("expected", "actual") 

25 )

28def test_shapes_match_so_the_differing_column_is_named(): 

29 # A datatype-only mismatch IS the two type IRIs, so the summary carries them 

30 # and the reader never has to reach the diff. 

31 df_diff = diff_of( 

32 {"month": ["2025-01"], "month_datatype": [XSD + "string"]}, 

33 {"month": ["2025-01"], "month_datatype": [XSD + "gYearMonth"]}, 

34 ) 

35 message = build_summary_message(1, 1, 1, 1, df_diff) 

36 

37 assert message == ( 

38 "Expected 1 row(s) and 1 column(s), got 1 row(s) and 1 column(s)" 

39 " — differs in: month (datatype: expected xsd:string, actual xsd:gYearMonth)" 

40 )
