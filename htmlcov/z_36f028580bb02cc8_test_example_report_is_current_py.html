1"""Golden test: the committed geography-example report is up to date. 

2 

3`docs/examples/geography-example/report/term-coverage-example.{md,ttl}` are 

4generated artifacts checked into the repo (so they render on GitHub and document 

5the feature). If a change alters the report, the committed copies must be 

6regenerated and committed. 

7 

8This regenerates both in the same environment they were committed from — from the 

9example directory, with the GitHub-Actions link/slug env removed so links are 

10relative and the run slug is 'local' — and fails (with the regenerate command) 

11if a committed copy is stale: 

12 

13- the **.md** is compared byte-for-byte (deterministic jinja output); 

14- the **.ttl** is compared as an RDF **graph** (triple sets), so it's immune to 

15 Turtle serialisation-order differences across rdflib versions. The graph has no 

16 blank nodes, so equal triple sets means an identical graph. 

17 

18Runs in CI (via `pytest test/`) and locally, so a stale report is caught before 

19merge. 

20""" 

21import difflib 

22from pathlib import Path 

23 

24import pytest 

25from rdflib import Graph, Namespace, URIRef, RDF 

26from rdflib.compare import to_isomorphic, graph_diff 

27 

28from mustrd.mustrdTestPlugin import MustrdTestPlugin 

29 

30COV = Namespace("https://mustrd.org/coverage/") 

31PROV = Namespace("http://www.w3.org/ns/prov#") 

32# Per-run provenance on the cov:CoverageRun node — different on every run, so 

33# excluded from the drift check (the golden test's whole point is structural). 

34_VOLATILE = (PROV.startedAtTime, PROV.endedAtTime, COV.gitRepository, 

35 COV.gitCommit, COV.gitCommitUrl, COV.ciRun) 

36 

37 

38def _canonical(graph): 

39 """Strip the run's per-run identity so two runs can be compared for drift: the 

40 unique run IRI (and every IRI minted under `.../run/<id>`) is rewritten to a 

41 fixed token, and the run's volatile provenance (time, SHA, commit/CI links) is 

42 dropped. Everything else — the coverage the report is built from — must match.""" 

43 run = graph.value(predicate=RDF.type, object=COV.CoverageRun) 

44 slug = str(run).rsplit("/", 1)[-1] if run is not None else None 

45 marker = f"/run/{slug}" if slug else None 

46 

47 def rw(t): 

48 if marker and isinstance(t, URIRef) and marker in str(t): 

49 return URIRef(str(t).replace(marker, "/run/__RUN__")) 

50 return t 

51 out = Graph() 

52 for s, p, o in graph: 

53 if run is not None and s == run and p in _VOLATILE: 

54 continue 

55 out.add((rw(s), rw(p), rw(o))) 

56 return out

58EXAMPLE = Path("docs/examples/geography-example").resolve() 

59COMMITTED_MD = Path("report/term-coverage-example.md") # relative to EXAMPLE 

60COMMITTED_TTL = Path("report/term-coverage-example.ttl") 

61 

62REGEN_CMD = ( 

63 "(cd docs/examples/geography-example && \\\n" 

64 " pytest . --mustrd --config=mustrd-config.ttl --term-coverage --cq \\\n" 

65 " --md=report/term-coverage-example.md \\\n" 

66 " --term-coverage-rdf=report/term-coverage-example.ttl)" 

67)

70def _stale(name, extra=""): 

71 return (f"{name} is out of date — regenerate and commit it:\n\n{REGEN_CMD}\n\n{extra}") 

72 

73 

74def _md_diff(generated, committed): 

75 return "".join(difflib.unified_diff( 

76 committed.splitlines(True), generated.splitlines(True), 

77 fromfile="committed", tofile="regenerated")) 

78 

79 

80def _graph_diff(generated, committed): 

81 """Isomorphism check via rdflib.compare; "" when the graphs are equal, else a 

82 human-readable diff of the differing triples. Both graphs are canonicalised 

83 first (per-run provenance removed) so only structural drift is reported.""" 

84 generated, committed = _canonical(generated), _canonical(committed) 

85 if to_isomorphic(generated) == to_isomorphic(committed): 

86 return "" 

87 _, only_committed, only_generated = graph_diff(to_isomorphic(committed), 

88 to_isomorphic(generated)) 

89 nm = committed.namespace_manager 

90 

91 def fmt(g): 

92 return "\n".join(" " + " ".join(t[i].n3(nm) for i in range(3)) 

93 for t in sorted(g, key=str)) 

94 parts = [] 

95 if only_committed: 

96 parts.append("in committed but not regenerated:\n" + fmt(only_committed)) 

97 if only_generated: 

98 parts.append("in regenerated but not committed:\n" + fmt(only_generated)) 

99 return "\n\n".join(parts)

102def test_committed_example_report_is_up_to_date(monkeypatch): 

103 # Reproduce the exact environment the committed files were generated in: from 

104 # the example dir, with the GitHub-Actions link/slug env removed. Write the 

105 # regenerated report into the same report/ directory (temp names) so its links 

106 # are relative to the same base as the committed copy. 

107 for var in ("GITHUB_ACTIONS", "GITHUB_SHA", "GITHUB_REPOSITORY", 

108 "GITHUB_SERVER_URL", "GITHUB_WORKSPACE"): 

109 monkeypatch.delenv(var, raising=False) 

110 monkeypatch.chdir(EXAMPLE) 

111 

112 md = COMMITTED_MD.parent / "__golden_check__.md" 

113 ttl = COMMITTED_MD.parent / "__golden_check__.ttl" 

114 plugin = MustrdTestPlugin(str(md), Path("mustrd-config.ttl"), None, 

115 term_coverage=True, cq=True, term_coverage_rdf=str(ttl)) 

116 try: 

117 pytest.main([".", "-p", "no:cacheprovider"], plugins=[plugin]) 

118 generated_md = md.read_text(encoding="utf-8") 

119 generated_graph = Graph().parse(ttl, format="turtle") 

120 finally: 

121 md.unlink(missing_ok=True) 

122 ttl.unlink(missing_ok=True) 

123 

124 committed_md = COMMITTED_MD.read_text(encoding="utf-8") 

125 if generated_md != committed_md: 

126 pytest.fail(_stale(COMMITTED_MD.name, _md_diff(generated_md, committed_md))) 

127 

128 committed_graph = Graph().parse(COMMITTED_TTL, format="turtle") 

129 diff = _graph_diff(generated_graph, committed_graph) 

130 if diff: 

131 pytest.fail(_stale(COMMITTED_TTL.name, diff))
