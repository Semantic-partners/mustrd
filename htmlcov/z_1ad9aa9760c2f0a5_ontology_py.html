1"""RDF / ontology introspection. 

2 

3Pure read layer beneath coverage.py and cq.py: loading ontology files, listing 

4their owl:Ontology metadata, deriving declared class/property terms, extracting 

5the terms an ABox or a SPARQL query references, namespace helpers, and prefix 

6shortening. Knows nothing about specs, coverage, or competency questions. 

7"""

21# Predicates checked, in order, for an ontology's human description. 

22DESCRIPTION_PREDICATES = (RDFS.comment, DCTERMS.description, DC.description, 

23 SKOS.definition, RDFS.label)

26# RDF serialisations recognised when scanning an ontology directory. 

27ONTOLOGY_SUFFIXES = {".ttl", ".trig", ".nt", ".nq", ".n3", ".jsonld", ".rdf", ".owl", ".xml"}

30# Namespaces whose terms are infrastructure, not "the ontology under test". 

31WELL_KNOWN = ( 

32 str(RDF), str(RDFS), str(OWL), str(XSD), 

33 "http://www.w3.org/2004/02/skos/core#", 

34 "http://www.w3.org/ns/shacl#", 

35 "https://mustrd.org/model/", 

36 "https://mustrd.org/competencyQuestion/", 

37 "http://purl.org/dc/elements/1.1/", 

38 "http://purl.org/dc/terms/", 

39 "http://www.w3.org/ns/prov#", 

40)

43CLASS_TYPES = (OWL.Class, RDFS.Class)

46PROPERTY_TYPES = frozenset(( 

47 RDF.Property, 

48 OWL.ObjectProperty, OWL.DatatypeProperty, OWL.AnnotationProperty, 

49 OWL.OntologyProperty, 

50 OWL.FunctionalProperty, OWL.InverseFunctionalProperty, 

51 OWL.SymmetricProperty, OWL.TransitiveProperty,

55# Property types that describe documentation/metadata rather than the domain 

56# vocabulary CQs are meant to exercise. A term declared *only* as one of these

58# than flagged as a gap. Mapped to the reason shown in the report. 

59METADATA_PROPERTY_TYPES = { 

60 OWL.AnnotationProperty: "annotation property", 

61 OWL.OntologyProperty: "ontology property",

68def wk_qname(uri) -> str: 

69 """Render an rdf/rdfs/owl IRI with its conventional prefix (owl:Class …).""" 

70 s = str(uri) 

71 for ns, pfx in _WELL_KNOWN_PREFIXES: 

72 if s.startswith(ns): 

73 return f"{pfx}:{s[len(ns):]}" 

74 return s

77def is_domain_term(uri) -> bool: 

78 return isinstance(uri, URIRef) and not any(str(uri).startswith(ns) for ns in WELL_KNOWN)

81def namespace(iri: str) -> str: 

82 """The namespace of an IRI — up to and including its last '#' or '/'.""" 

83 for sep in ("#", "/"): 

84 idx = iri.rfind(sep) 

85 if idx != -1: 

86 return iri[:idx + 1] 

87 return iri

90def expand_ontology_files(paths) -> list: 

91 """Expand a list of file/directory paths into a sorted list of RDF files. 

92 

93 Files are kept as-is; directories are scanned recursively for files with a 

94 recognised RDF suffix. Non-existent paths are skipped with a warning. 

95 """ 

96 files = [] 

97 for p in paths: 

98 p = Path(p) 

99 if p.is_dir(): 

100 files.extend(sorted(f for f in p.rglob("*") 

101 if f.is_file() and f.suffix.lower() in ONTOLOGY_SUFFIXES)) 

102 elif p.is_file(): 

103 files.append(p) 

104 else: 

105 log.warning(f"hasOntologyPath does not exist, skipping: {p}") 

106 # de-duplicate while preserving order 

107 seen, unique = set(), [] 

108 for f in files: 

109 rp = f.resolve() 

110 if rp not in seen: 

111 seen.add(rp) 

112 unique.append(f) 

113 return unique

116def ontology_report(paths, link_base=None, href=None) -> list: 

117 """Per-file summary of the ontologies under `paths`, for the report header. 

118 

119 Each entry: {path, url, uri, description} — the file link (its href, see 

120 `_source_link`), the owl:Ontology IRI declared in that file (if any), and its 

121 description (rdfs:comment / dcterms:description / … ). A file with no 

122 owl:Ontology still appears (uri/description None); a file declaring several 

123 yields one row each. `href`, if given, is a callable path->url that overrides 

124 the default `link_base`-relative href (e.g. absolute GitHub URLs in CI). 

125 """ 

126 rows = [] 

127 for f in expand_ontology_files(paths): 

128 link = _source_link(f, link_base, href) 

129 g = Graph() 

130 try: 

131 g.parse(str(f)) 

132 except Exception as e: 

133 log.warning(f"Could not parse ontology file {f}: {e}") 

134 rows.append({**link, "uri": None, "description": None}) 

135 continue 

136 ontologies = sorted(str(s) for s in g.subjects(RDF.type, OWL.Ontology)) 

137 if not ontologies: 

138 rows.append({**link, "uri": None, "description": None}) 

139 for uri in ontologies: 

140 rows.append({**link, "uri": uri, 

141 "description": _first_literal(g, URIRef(uri), DESCRIPTION_PREDICATES)}) 

142 return rows

145def _first_literal(graph: Graph, subject, predicates) -> Optional[str]: 

146 for p in predicates: 

147 val = graph.value(subject=subject, predicate=p) 

148 if val is not None: 

149 return str(val) 

150 return None

153def load_ontology(paths) -> Optional[Graph]: 

154 """Parse every ontology file under `paths` (files or dirs) into one graph.""" 

155 files = expand_ontology_files(paths) 

156 if not files: 

157 return None 

158 g = Graph() 

159 for f in files: 

160 try: 

161 g.parse(str(f)) 

162 except Exception as e: 

163 log.warning(f"Could not parse ontology file {f}: {e}") 

164 return g

167def declared_terms(graph: Graph) -> dict: 

168 """Map each declared class/property IRI in the graph to 'class' or 'property'. 

169 

170 Restricted to non-well-known namespaces (the ontology under test, not the 

171 RDF/RDFS/OWL/SKOS vocabulary it is written in). 

172 """ 

173 terms = {} 

174 for t in CLASS_TYPES: 

175 for s in graph.subjects(RDF.type, t): 

176 if is_domain_term(s): 

177 terms.setdefault(str(s), "class") 

178 for t in PROPERTY_TYPES: 

179 for s in graph.subjects(RDF.type, t): 

180 if is_domain_term(s): 

181 terms[str(s)] = "property" # a property label wins over a class collision 

182 return terms

185def metadata_terms(graph: Graph) -> dict: 

186 """Domain terms declared *only* as annotation/ontology properties. 

187 

188 Maps each such IRI to a reason label ("annotation property" / 

189 "ontology property"). These are documentation/metadata vocabulary, not the 

190 substantive classes and properties CQs exercise, so coverage reports an 

191 unused one as a schema term rather than a gap. A term also declared as a 

192 class or a substantive property is excluded — it is not "just metadata". 

193 """ 

194 meta = {} 

195 for typ, label in METADATA_PROPERTY_TYPES.items(): 

196 for s in graph.subjects(RDF.type, typ): 

197 if is_domain_term(s): 

198 meta.setdefault(str(s), label) 

199 substantive = set() 

200 for typ in CLASS_TYPES: 

201 substantive |= {str(s) for s in graph.subjects(RDF.type, typ)} 

202 for typ in PROPERTY_TYPES - set(METADATA_PROPERTY_TYPES): 

203 substantive |= {str(s) for s in graph.subjects(RDF.type, typ)} 

204 return {iri: label for iri, label in meta.items() if iri not in substantive}

207def _collect_uris(root) -> set: 

208 """Every URIRef reachable from a parsed-algebra node, walked iteratively. 

209 

210 Descends dicts, sequences and objects' ``__dict__``; a seen-set on object 

211 identity guards against cycles. 

212 """ 

213 found, seen, stack = set(), set(), [root] 

214 while stack: 

215 obj = stack.pop() 

216 if id(obj) in seen: 

217 continue 

218 seen.add(id(obj)) 

219 if isinstance(obj, URIRef): 

220 found.add(str(obj)) 

221 elif isinstance(obj, dict): 

222 stack.extend(obj.values()) 

223 elif isinstance(obj, (list, tuple, set)): 

224 stack.extend(obj) 

225 elif hasattr(obj, "__dict__"): 

226 stack.extend(vars(obj).values()) 

227 return found

230def query_uris(query_text: str) -> set: 

231 """Every IRI referenced in a query's parsed algebra (ignores comments). 

232 

233 Handles SELECT/CONSTRUCT/ASK/DESCRIBE and, as a fallback, SPARQL Update. 

234 """ 

235 try: 

236 algebra = prepareQuery(query_text).algebra 

237 except Exception as query_exc: 

238 try: 

239 from rdflib.plugins.sparql.parser import parseUpdate 

240 from rdflib.plugins.sparql.algebra import translateUpdate 

241 algebra = translateUpdate(parseUpdate(query_text)) 

242 except Exception as update_exc: 

243 log.debug("query_uris: could not parse as query (%s) nor update (%s); " 

244 "extracting no query terms from: %s", 

245 query_exc, update_exc, query_text) 

246 return set() 

247 return _collect_uris(algebra)

250def abox_terms(graph: Graph) -> set: 

251 """Terms USED by instance data: rdf:type objects + asserted predicates.""" 

252 used = {str(o) for o in graph.objects(None, RDF.type) if isinstance(o, URIRef)} 

253 used |= {str(p) for p in set(graph.predicates()) if isinstance(p, URIRef)} 

254 return used

260def shortener(graphs, query_texts=()): 

261 """Build a prefix map -> function turning an IRI into a qname. 

262 

263 Sources both the given graphs and the `PREFIX` declarations in the query 

264 text. mustrd's given graph often loses author prefixes (rdflib 

265 auto-generates `ns1`, `ns2`, ... on collisions), whereas the SPARQL 

266 `PREFIX ex: <...>` lines are author-chosen and clean. Per namespace we pick 

267 the cleanest prefix: prefer one not ending in a digit, then the shortest. 

268 """ 

269 candidates = {} # namespace -> set of prefixes 

270 for g in graphs: 

271 for prefix, ns in g.namespaces(): 

272 if prefix: # skip the default (empty) prefix 

273 candidates.setdefault(str(ns), set()).add(prefix) 

274 for text in query_texts: 

275 for prefix, ns in _PREFIX_RE.findall(text or ""): 

276 candidates.setdefault(ns, set()).add(prefix) 

277 

278 def best(prefixes): 

279 return sorted(prefixes, key=lambda p: (p[-1].isdigit(), len(p), p))[0] 

280 

281 ns_to_prefix = {ns: best(pfx) for ns, pfx in candidates.items()} 

282 # longest namespace first, so the most specific binding wins 

283 ordered = sorted(ns_to_prefix.items(), key=lambda kv: len(kv[0]), reverse=True) 

284 

285 def short(uri: str) -> str: 

286 for ns, prefix in ordered: 

287 if uri.startswith(ns): 

288 return f"{prefix}:{uri[len(ns):]}" 

289 return uri 

290 

291 return short

294def _source_link(p, link_base=None, href=None) -> dict: 

295 """Display label + a link href. 

296 

297 When `href` (a path->url callable) is given it decides the url — used to emit 

298 absolute GitHub URLs in CI. Otherwise the url is RELATIVE to `link_base` 

299 (Markdown previewers like VS Code block absolute `file://` links, so a 

300 relative href resolves against the report's own location): the report file's 

301 directory for an `--md` file, or the cwd for terminal output. The label stays 

302 relative to the cwd for readability. 

303 """ 

304 p = Path(p) 

305 try: 

306 label = os.path.relpath(p) 

307 except ValueError: # e.g. different drive on Windows 

308 label = str(p) 

309 if href is not None: 

310 url = href(p) 

311 else: 

312 base = Path(link_base) if link_base is not None else Path.cwd() 

313 try: 

314 url = os.path.relpath(p, base) 

315 except ValueError: 

316 url = str(p) 

317 return {"path": label, "url": url}
