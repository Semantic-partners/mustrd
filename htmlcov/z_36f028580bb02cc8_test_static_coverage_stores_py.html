1"""Static coverage contacts no store and reads no credentials. 

2 

3`mustrd coverage`, `collect_specs` and `static_coverage` build every spec and run 

4none. They used to resolve each suite's triple stores anyway: with the store 

5config missing, or present without a token, `filterOnTripleStore` then removed 

6the rdflib fallback, the suite had no stores, and every spec was dropped without 

7a word. These tests pin the store-free path and the count of what was left out. 

8"""

87def _suite(root, store_config=None, invalid=False): 

88 (root / "specs").mkdir() 

89 (root / "data").mkdir() 

90 (root / "specs" / "inherited.mustrd.ttl").write_text(SPEC, encoding="utf-8") 

91 if invalid: 

92 (root / "specs" / "no-when.mustrd.ttl").write_text(INVALID_SPEC, encoding="utf-8") 

93 (root / "onto.ttl").write_text(ONTO, encoding="utf-8") 

94 (root / "mustrd-config.ttl").write_text(CONFIG, encoding="utf-8") 

95 if store_config is not None: 

96 (root / "triplestore-config.ttl").write_text(store_config, encoding="utf-8") 

97 return root / "mustrd-config.ttl"
