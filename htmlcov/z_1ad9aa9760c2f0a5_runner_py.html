1"""Run mustrd specs without pytest — the orchestration the `mustrd` CLI needs. 

2 

3Mirrors what the pytest plugin does at collection + run time, reusing the same 

4mustrd.py pipeline (validate_specs -> get_specs -> run_spec) and the same 

5`mustrd.config` TestConfig, so the CLI and the plugin execute identical specs. 

6The pytest plugin delegates its spec-generation to the functions here, keeping a 

7single source of truth. 

8"""

35def resolve_triple_stores(test_config, secrets=None): 

36 """The triple stores a TestConfig runs against — from its 

37 triplestoreSpecPath, or embedded rdflib, then filtered by 

38 filterOnTripleStore. (Extracted from the plugin's 

39 get_triple_stores_from_file so the CLI shares it.)""" 

40 if test_config.triplestore_spec_path: 

41 try: 

42 triple_stores = get_triple_stores( 

43 get_triple_store_graph(test_config.triplestore_spec_path), 

44 get_credentials(test_config.triplestore_spec_path, secrets), 

45 ) 

46 except Exception as e: 

47 print( 

48 f"""Triplestore configuration parsing failed {test_config.triplestore_spec_path}. 

49 Only rdflib will be executed""", 

50 e, 

51 ) 

52 triple_stores = [dict(_RDFLIB_STORE)] 

53 else: 

54 logger.debug("No triple store configuration required: using embedded rdflib") 

55 triple_stores = [dict(_RDFLIB_STORE)] 

56 

57 if test_config.filter_on_tripleStore: 

58 triple_stores = list( 

59 filter( 

60 lambda triple_store: ( 

61 triple_store["uri"] in test_config.filter_on_tripleStore 

62 ), 

63 triple_stores, 

64 ) 

65 ) 

66 return triple_stores

69def generate_specs(config, triple_stores, file_name="*", selected_tests=None, 

70 ignore_focus=False): 

71 """Validate + build the Specifications for one TestConfig across the given 

72 triple stores. Returns runnable specs + skipped/invalid SpecResults. 

73 (Extracted from the plugin's generate_tests_for_config.)""" 

74 shacl_graph = Graph().parse(Path(os.path.join(mustrd_root, "model/mustrdShapes.ttl"))) 

75 ont_graph = Graph().parse(Path(os.path.join(mustrd_root, "model/ontology.ttl"))) 

76 

77 valid_spec_uris, spec_graph, invalid_specs = validate_specs( 

78 config, triple_stores, shacl_graph, ont_graph, file_name or "*", 

79 selected_test_files=selected_tests or [], ignore_focus=ignore_focus, 

80 ) 

81 specs, skipped_spec_results = get_specs( 

82 valid_spec_uris, spec_graph, triple_stores, config 

83 ) 

84 return specs, skipped_spec_results + invalid_specs

87def _outcome(result): 

88 """'passed' / 'failed' for a SpecResult (mirrors the plugin's pytest 

89 outcome for coverage purposes).""" 

90 return "passed" if isinstance(result, (SpecPassed, SpecPassedWithWarning)) else "failed"

93def triple_store_name(spec): 

94 ts = getattr(spec, "triple_store", None) 

95 if isinstance(ts, dict): 

96 return str(ts.get("uri") or ts.get("type") or "rdflib").split("/")[-1].split("#")[-1] 

97 return str(ts).split("/")[-1].split("#")[-1] if ts else "rdflib"

107def static_triple_stores(test_config): 

108 """The stores a TestConfig is measured against when nothing runs. 

109 

110 Static coverage never contacts a store, so it reads no connection config and 

111 no credentials: not triplestoreSpecPath, not a `_secrets` file, not 

112 `:tokenEnv` / `:tokenFile`. One placeholder per `filterOnTripleStore` value, 

113 typed rdflib so nothing treats it as a live server, and named after the 

114 filter so test names still read `spec@CiscStardog`. With no filter, the 

115 rdflib store. Never empty, so a suite can never resolve to no store and lose 

116 its specs silently. 

117 """ 

118 if test_config.filter_on_tripleStore: 

119 return [{"type": TRIPLESTORE.RdfLib, "uri": URIRef(str(f))} 

120 for f in test_config.filter_on_tripleStore] 

121 return [dict(_RDFLIB_STORE)]

124def _built_specs(config_paths, stores_for, selected_tests=None, ignore_focus=False): 

125 """Build (but do not run) every spec in the configs, against the stores 

126 `stores_for(test_config)` returns. Returns a (test_config, specs, skipped) 

127 triple per TestConfig, and the hasSpecPath dirs.""" 

128 test_configs = [tc for f in _config_files(config_paths) for tc in parse_config(f)] 

129 per_config = [] 

130 for test_config in test_configs: 

131 triple_stores = stores_for(test_config) 

132 # validate_specs / get_specs expect a run_config dict, not the TestConfig 

133 # dataclass — the same shape the plugin builds during collection. 

134 run_cfg = {"spec_path": test_config.spec_path, 

135 "data_path": test_config.data_path} 

136 specs, skipped = generate_specs(run_cfg, triple_stores, 

137 selected_tests=selected_tests, 

138 ignore_focus=ignore_focus) 

139 per_config.append((test_config, specs, skipped)) 

140 return per_config, [tc.spec_path for tc in test_configs if tc.spec_path]

143@dataclass 

144class StaticSpecs: 

145 """What static collection found: the coverage inputs, and every spec left 

146 out with the reason. 

147 

148 `left_out` is a list of {"name", "reason"}: specs that failed validation or 

149 could not be built (e.g. an Anzo-only source with no Anzo to read from). 

150 """ 

151 all_specs: list = field(default_factory=list) 

152 spec_by_uri: dict = field(default_factory=dict) 

153 spec_paths: list = field(default_factory=list) 

154 left_out: list = field(default_factory=list) 

155 

156 def summary(self): 

157 # Distinct specs, not spec×store: a suite filtered to two stores builds 

158 # each spec twice, and the count is of specs. 

159 n = len({s.get("uri") or s.get("test_name") for s in self.all_specs}) 

160 m = len(self.left_out) 

161 return f"{n} spec{'' if n == 1 else 's'} measured, {m} not counted"

164def collect_static(config_paths, selected_tests=None, ignore_focus=False) -> StaticSpecs: 

165 """Build every spec in the configs and run none, against placeholder stores 

166 (see `static_triple_stores`): no network, no credentials. 

167 

168 Every spec that builds is credited as passing: coverage then reads what the 

169 specs say (their given data and queries), not whether they hold against a 

170 store. Specs that fail validation or cannot be built go in `left_out`. 

171 """ 

172 per_config, spec_paths = _built_specs(config_paths, static_triple_stores, 

173 selected_tests, ignore_focus) 

174 out = StaticSpecs(spec_paths=spec_paths) 

175 for _tc, specs, skipped in per_config: 

176 for spec in specs: 

177 test_name = f"{getattr(spec, 'spec_file_name', spec.spec_uri)}@{triple_store_name(spec)}" 

178 cspec = coverage_spec(spec, "passed", test_name) 

179 cspec["test_name"] = test_name 

180 out.all_specs.append(cspec) 

181 if cspec.get("uri"): 

182 out.spec_by_uri[cspec["uri"]] = cspec 

183 # One entry per spec, not per store: an invalid spec is recorded once for 

184 # each store it was tried on, and carries only the store's type. 

185 seen = set() 

186 for sk in skipped: 

187 uri = getattr(sk, "spec_uri", None) 

188 name = getattr(sk, "spec_file_name", None) or (str(uri) if uri else "unknown") 

189 reason = (getattr(sk, "message", None) or type(sk).__name__).strip() 

190 reason = reason.splitlines()[0] if reason else "" 

191 if (str(uri), name) in seen: 

192 continue 

193 seen.add((str(uri), name)) 

194 out.left_out.append({"name": name, "reason": reason}) 

195 return out

198def collect_specs(config_paths, secrets=None, selected_tests=None, ignore_focus=False): 

199 """The coverage inputs for the configs' specs WITHOUT running any of them. 

200 

201 (all_specs, spec_by_uri, spec_paths) 

202 

203 See `collect_static`, which also reports the specs left out. `secrets` is 

204 accepted for compatibility and ignored: nothing here connects to a store. 

205 """ 

206 found = collect_static(config_paths, selected_tests, ignore_focus) 

207 return found.all_specs, found.spec_by_uri, found.spec_paths

210def static_coverage(config_paths, opts=None, secrets=None, selected_tests=None, 

211 ignore_focus=False, terminal_writer=None, collected=None): 

212 """Term coverage for one or more MustrdTest configs, WITHOUT running the tests. 

213 

214 Builds the specs, credits each one as passing (see `collect_static`), and 

215 writes whatever `opts` (a `mustrd.reporting.ReportOptions`) asks for: --md, 

216 the coverage graph, the viewer. With no `opts`, computes term coverage and the 

217 competency-question overlay and writes nothing. Ontology paths default to the 

218 configs' hasOntologyPath. 

219 

220 No store is contacted and no credentials are read. `secrets` is accepted for 

221 compatibility and ignored. Pass `collected` (from `collect_static`) to reuse 

222 a collection, e.g. to report its counts; specs left out are logged as a 

223 warning either way. 

224 

225 Returns `coverage`, the template context dict (`covered`, `queried`, `pct`, 

226 `queried_pct`, ...), or None when no ontology terms are declared. 

227 """ 

228 from mustrd.reporting import (ReportOptions, collect_cq_defs, produce_report, 

229 wants_cq) 

230 opts = opts or ReportOptions(term_coverage=True, cq=True) 

231 if not opts.ontology_paths: 

232 opts.ontology_paths = tuple(ontology_paths_from_config(config_paths)) 

233 found = collected or collect_static(config_paths, selected_tests, ignore_focus) 

234 for lo in found.left_out: 

235 logger.warning(f"not counted in coverage: {lo['name']}: {lo['reason']}") 

236 cq_defs = collect_cq_defs(found.spec_paths, found.spec_by_uri) if wants_cq(opts) else [] 

237 coverage, _ontology_graph, _graph = produce_report( 

238 found.all_specs, cq_defs, [], True, opts, terminal_writer=terminal_writer) 

239 return coverage

242def run_config(config_paths, secrets=None, selected_tests=None, ignore_focus=False, 

243 verbose=False, review=False): 

244 """Run every spec in one or more MustrdTest config files and return the plain-data inputs the 

245 reporting library consumes: 

246 

247 (results, all_specs, spec_by_uri, test_results, run_results, spec_paths) 

248 

249 - results: the SpecResults (passed/failed + skipped/invalid), for review. 

250 - all_specs / spec_by_uri: coverage spec-dicts (as the plugin builds them). 

251 - test_results: a TestResult per run spec, for the plain --md ResultList. 

252 - run_results: a RunResult per test (incl. skipped/invalid), for the results graph. 

253 - spec_paths: hasSpecPath dirs, for competency-question discovery. 

254 """ 

255 results, all_specs, spec_by_uri, test_results, run_results = [], [], {}, [], [] 

256 per_config, spec_paths = _built_specs( 

257 config_paths, lambda tc: resolve_triple_stores(tc, secrets), 

258 selected_tests, ignore_focus) 

259 

260 for _tc, specs, skipped in per_config: 

261 for spec in specs: 

262 ts = triple_store_name(spec) 

263 t0 = time.perf_counter() 

264 result = run_spec(spec) 

265 duration = time.perf_counter() - t0 

266 results.append(result) 

267 outcome = _outcome(result) 

268 test_name = f"{getattr(spec, 'spec_file_name', spec.spec_uri)}@{ts}" 

269 test_results.append(TestResult(test_name, ts, "mustrd", outcome, True)) 

270 uri = getattr(spec, "spec_uri", None) 

271 src = getattr(spec, "spec_source_file", None) 

272 run_results.append(RunResult( 

273 status=outcome, test_type="mustrd", module="mustrd", 

274 class_name=ts, test_name=test_name, 

275 spec_uri=str(uri) if uri is not None else None, 

276 spec_file_name=getattr(spec, "spec_file_name", None), 

277 source_file=str(src) if src is not None else None, 

278 duration=duration)) 

279 cspec = coverage_spec(spec, outcome, test_name) 

280 all_specs.append(cspec) 

281 if cspec.get("uri"): 

282 spec_by_uri[cspec["uri"]] = cspec 

283 # Skipped/invalid specs still count as results for review, and appear as 

284 # skipped in the results graph. 

285 results.extend(skipped) 

286 for sk in skipped: 

287 uri = getattr(sk, "spec_uri", None) 

288 ts = str(getattr(sk, "triple_store", "")).split("/")[-1].split("#")[-1] 

289 src = getattr(sk, "spec_source_file", None) 

290 name = getattr(sk, "spec_file_name", None) or (str(uri) if uri else "unknown") 

291 run_results.append(RunResult( 

292 status="skipped", test_type="mustrd", module="mustrd", 

293 class_name=ts, test_name=f"{name}@{ts}", 

294 spec_uri=str(uri) if uri is not None else None, 

295 spec_file_name=getattr(sk, "spec_file_name", None), 

296 source_file=str(src) if src is not None else None)) 

297 

298 if review: 

299 review_results(results, verbose) 

300 return results, all_specs, spec_by_uri, test_results, run_results, spec_paths
