1"""Run mustrd specs without pytest — the orchestration the `mustrd` CLI needs. 

2 

3Mirrors what the pytest plugin does at collection + run time, reusing the same 

4mustrd.py pipeline (validate_specs -> get_specs -> run_spec) and the same 

5`mustrd.config` TestConfig, so the CLI and the plugin execute identical specs. 

6The pytest plugin delegates its spec-generation to the functions here, keeping a 

7single source of truth. 

8""" 

9import logging 

10import os 

11import time 

12from pathlib import Path

14from rdflib import Graph

17 validate_specs, get_specs, run_spec, review_results, 

18 SpecPassed, SpecPassedWithWarning, 

19 get_triple_store_graph, get_triple_stores, 

20) 

21from mustrd.config import parse_config 

22from mustrd.namespace import TRIPLESTORE 

23from mustrd.reporting import coverage_spec 

24from mustrd.results_rdf import RunResult 

25from mustrd.TestResult import TestResult

29mustrd_root = get_mustrd_root()

34def resolve_triple_stores(test_config, secrets=None): 

35 """The triple stores a TestConfig runs against — from its 

36 triplestoreSpecPath, or embedded rdflib, then filtered by 

37 filterOnTripleStore. (Extracted from the plugin's 

38 get_triple_stores_from_file so the CLI shares it.)""" 

39 if test_config.triplestore_spec_path: 

40 try: 

41 triple_stores = get_triple_stores( 

42 get_triple_store_graph(test_config.triplestore_spec_path, secrets) 

43 ) 

44 except Exception as e: 

45 print( 

46 f"""Triplestore configuration parsing failed {test_config.triplestore_spec_path}. 

47 Only rdflib will be executed""", 

48 e, 

49 ) 

50 triple_stores = [dict(_RDFLIB_STORE)] 

51 else: 

52 logger.debug("No triple store configuration required: using embedded rdflib") 

53 triple_stores = [dict(_RDFLIB_STORE)] 

54 

55 if test_config.filter_on_tripleStore: 

56 triple_stores = list( 

57 filter( 

58 lambda triple_store: ( 

59 triple_store["uri"] in test_config.filter_on_tripleStore 

60 ), 

61 triple_stores, 

62 ) 

63 ) 

64 return triple_stores

67def generate_specs(config, triple_stores, file_name="*", selected_tests=None, 

68 ignore_focus=False): 

69 """Validate + build the Specifications for one TestConfig across the given 

70 triple stores. Returns runnable specs + skipped/invalid SpecResults. 

71 (Extracted from the plugin's generate_tests_for_config.)""" 

72 shacl_graph = Graph().parse(Path(os.path.join(mustrd_root, "model/mustrdShapes.ttl"))) 

73 ont_graph = Graph().parse(Path(os.path.join(mustrd_root, "model/ontology.ttl"))) 

74 

75 valid_spec_uris, spec_graph, invalid_specs = validate_specs( 

76 config, triple_stores, shacl_graph, ont_graph, file_name or "*", 

77 selected_test_files=selected_tests or [], ignore_focus=ignore_focus, 

78 ) 

79 specs, skipped_spec_results = get_specs( 

80 valid_spec_uris, spec_graph, triple_stores, config 

81 ) 

82 return specs, skipped_spec_results + invalid_specs

85def _outcome(result): 

86 """'passed' / 'failed' for a SpecResult (mirrors the plugin's pytest 

87 outcome for coverage purposes).""" 

88 return "passed" if isinstance(result, (SpecPassed, SpecPassedWithWarning)) else "failed"

91def _triple_store_name(spec): 

92 ts = getattr(spec, "triple_store", None) 

93 if isinstance(ts, dict): 

94 return str(ts.get("uri") or ts.get("type") or "rdflib").split("/")[-1].split("#")[-1] 

95 return str(ts).split("/")[-1].split("#")[-1] if ts else "rdflib"

98def run_config(config_path, secrets=None, selected_tests=None, ignore_focus=False, 

99 verbose=False, review=False): 

100 """Run every spec in a MustrdTest config and return the plain-data inputs the 

101 reporting library consumes: 

102 

103 (results, all_specs, spec_by_uri, test_results, run_results, spec_paths) 

104 

105 - results: the SpecResults (passed/failed + skipped/invalid), for review. 

106 - all_specs / spec_by_uri: coverage spec-dicts (as the plugin builds them). 

107 - test_results: a TestResult per run spec, for the plain --md ResultList. 

108 - run_results: a RunResult per test (incl. skipped/invalid), for the results graph. 

109 - spec_paths: hasSpecPath dirs, for competency-question discovery. 

110 """ 

111 test_configs = parse_config(Path(config_path)) 

112 results, all_specs, spec_by_uri, test_results, run_results = [], [], {}, [], [] 

113 spec_paths = [tc.spec_path for tc in test_configs if tc.spec_path] 

114 

115 for test_config in test_configs: 

116 triple_stores = resolve_triple_stores(test_config, secrets) 

117 # validate_specs / get_specs expect a run_config dict, not the TestConfig 

118 # dataclass — the same shape the plugin builds during collection. 

119 run_cfg = {"spec_path": test_config.spec_path, 

120 "data_path": test_config.data_path} 

121 specs, skipped = generate_specs(run_cfg, triple_stores, 

122 selected_tests=selected_tests, 

123 ignore_focus=ignore_focus) 

124 for spec in specs: 

125 ts = _triple_store_name(spec) 

126 t0 = time.perf_counter() 

127 result = run_spec(spec) 

128 duration = time.perf_counter() - t0 

129 results.append(result) 

130 outcome = _outcome(result) 

131 test_name = f"{getattr(spec, 'spec_file_name', spec.spec_uri)}@{ts}" 

132 test_results.append(TestResult(test_name, ts, "mustrd", outcome, True)) 

133 uri = getattr(spec, "spec_uri", None) 

134 src = getattr(spec, "spec_source_file", None) 

135 run_results.append(RunResult( 

136 status=outcome, test_type="mustrd", module="mustrd", 

137 class_name=ts, test_name=test_name, 

138 spec_uri=str(uri) if uri is not None else None, 

139 spec_file_name=getattr(spec, "spec_file_name", None), 

140 source_file=str(src) if src is not None else None, 

141 duration=duration)) 

142 cspec = coverage_spec(spec, outcome, test_name) 

143 all_specs.append(cspec) 

144 if cspec.get("uri"): 

145 spec_by_uri[cspec["uri"]] = cspec 

146 # Skipped/invalid specs still count as results for review, and appear as 

147 # skipped in the results graph. 

148 results.extend(skipped) 

149 for sk in skipped: 

150 uri = getattr(sk, "spec_uri", None) 

151 ts = str(getattr(sk, "triple_store", "")).split("/")[-1].split("#")[-1] 

152 src = getattr(sk, "spec_source_file", None) 

153 name = getattr(sk, "spec_file_name", None) or (str(uri) if uri else "unknown") 

154 run_results.append(RunResult( 

155 status="skipped", test_type="mustrd", module="mustrd", 

156 class_name=ts, test_name=f"{name}@{ts}", 

157 spec_uri=str(uri) if uri is not None else None, 

158 spec_file_name=getattr(sk, "spec_file_name", None), 

159 source_file=str(src) if src is not None else None)) 

160 

161 if review: 

162 review_results(results, verbose) 

163 return results, all_specs, spec_by_uri, test_results, run_results, spec_paths
