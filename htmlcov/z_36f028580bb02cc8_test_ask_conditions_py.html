129def test_an_installed_rewriter_is_used(monkeypatch, spec_dir, capsys): 

130 calls = [] 

131 

132 def rewriter(select, conditions, min_rows, max_rows): 

133 calls.append(select) 

134 # Not FILTER(false): rdflib 7.6 answers `ASK { FILTER(false) }` with true. 

135 return [AskCheck("from the installed rewriter", "ASK { FILTER(1 = 2) }")] 

136 

137 entry_point = SimpleNamespace(load=lambda: rewriter) 

138 monkeypatch.setattr(ask_rewrite, "available_rewriters", 

139 lambda: {"subquery": None, "custom": entry_point}) 

140 try: 

141 assert main(["run", "--config", str(spec_dir / "config.ttl"), "-v", 

142 "--ask-rewriter", "custom"]) == 1 

143 finally: 

144 set_rewriter(None) 

145 assert calls == [PEOPLE] 

146 captured = capsys.readouterr() 

147 assert "Expected from the installed rewriter" in captured.out + captured.err
